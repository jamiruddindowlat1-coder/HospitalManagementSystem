

        [HttpGet]
        public async Task<ActionResult<IEnumerable<MedicalRecord>>> GetMedicalRecords()
        {
            return await _context.MedicalRecords

                .Include(m => m.Appointment)
                    .ThenInclude(a => a.Patient)

                .Include(m => m.Appointment)
                    .ThenInclude(a => a.Doctor)

                .ToListAsync();
        }
